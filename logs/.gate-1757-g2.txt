run_case() {
  local label=$1
  shift
  local dir="$work/$label"
  mkdir -p "$dir"
  printf 'START %s %s\n' "$(date --iso-8601=seconds)" "$label"
  mpirun --oversubscribe -np 16 --bind-to none \
    --map-by ppr:1:numa --allow-run-as-root \
    python -B "$root/pyqcu/testing/simple/kupl_multigrid.py" \
    --lat 24 24 24 64 --grid 1 1 2 8 \
    --gauge-slab-dir "$gauge_slabs" --source-slab-dir "$source_slabs" \
    --mass -0.277 --csw 1.160920226 --atol 1e-4 --max-iter 50 \
    --dtype c64 "$@" \
    --gauge "$case_dir/gauge_24x24x24x64_c64.h5" \
    --source "$case_dir/source_24x24x24x64_c64.h5" \
    --report "$dir/report.json" --output "$dir/solution.h5" \
    --skip-independent-residual >"$dir/stdout.txt" 2>"$dir/stderr.txt"
  local rc=$?
  printf 'END %s %s rc=%s\n' "$(date --iso-8601=seconds)" "$label" "$rc"
  return "$rc"
}

run_bicg() {
  unset KUPL_REQUIRE_MG_CORRECTION
  run_case "$1" --levels 1 --restart 0
}

run_mg() {
  export KUPL_REQUIRE_MG_CORRECTION=1
  run_case "$1" --levels 2 --restart 10 --mg-grid 2 2 2 1 \
    --dof 12 4 --nv-iters 2 --coarse-tol-factor 10 \
    --coarse-max-iter 20 --coarse-cache "$cache"
}
