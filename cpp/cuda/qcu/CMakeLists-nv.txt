cmake_minimum_required(VERSION 3.8)
project(qcu LANGUAGES C CXX CUDA)
set(CMAKE_CXX_STANDARD 11)
option(QCU_ENABLE_NVSHMEM "Enable the optional NVSHMEM coarse-grid transport" OFF)
message(STATUS "PATH is: $ENV{PATH}")
message(STATUS "LD_LIBRARY_PATH is: $ENV{LD_LIBRARY_PATH}")
# Architecture precedence: explicit QCU_CUDA_ARCHITECTURES, then the
# sm_arch detected by env.sh, then the historical sm60 compatibility image.
# make.sh sources env.sh before invoking CMake, so normal builds automatically
# target the installed GPU instead of being locked to sm60.
set(QCU_CUDA_ARCHITECTURES "$ENV{QCU_CUDA_ARCHITECTURES}")
if(QCU_CUDA_ARCHITECTURES STREQUAL "")
  if(DEFINED ENV{sm_arch} AND NOT "$ENV{sm_arch}" STREQUAL "")
    set(QCU_CUDA_ARCHITECTURES "$ENV{sm_arch}")
    message(STATUS "QCU CUDA architecture auto-detected from env.sh: sm_${QCU_CUDA_ARCHITECTURES}")
  else()
    set(QCU_CUDA_ARCHITECTURES "60-real;60-virtual")
    message(STATUS "QCU CUDA architecture fallback: ${QCU_CUDA_ARCHITECTURES}")
  endif()
endif()
set(CMAKE_CUDA_ARCHITECTURES "${QCU_CUDA_ARCHITECTURES}")
set(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} -O3")
message(STATUS "QCU CUDA architectures: ${QCU_CUDA_ARCHITECTURES}")
set(MPI_INCLUDE_PATH $ENV{mpi_home}/include)
set(MPI_CXX_LIBRARIES $ENV{mpi_home}/lib/libmpi.so)
include(GNUInstallDirs)
include_directories(${LD_LIBRARY_PATH})
include_directories(${MPI_INCLUDE_PATH})
include_directories(./include)
aux_source_directory(./src SOURCE_FILE)
add_library(qcu SHARED ${SOURCE_FILE})
find_package(MPI REQUIRED)
find_package(CUDA REQUIRED)
target_link_libraries(qcu ${MPI_C_LIBRARIES})
target_link_libraries(qcu ${CUDA_LIBRARIES})
target_link_libraries(qcu curand)
target_link_libraries(qcu cublas)

# NVSHMEM is deliberately opt-in.  A normal CUDA/MPI build must not need the
# headers or library, while an explicit request must fail during configure
# instead of producing a binary whose optional transport silently disappears.
if(QCU_ENABLE_NVSHMEM)
  find_path(QCU_NVSHMEM_INCLUDE_DIR
            NAMES nvshmem.h
            HINTS $ENV{NVSHMEM_HOME} $ENV{NVSHMEM_ROOT}
            PATH_SUFFIXES include)
  find_path(QCU_NVSHMEMX_INCLUDE_DIR
            NAMES nvshmemx.h
            HINTS $ENV{NVSHMEM_HOME} $ENV{NVSHMEM_ROOT}
            PATH_SUFFIXES include)
  find_library(QCU_NVSHMEM_LIBRARY
               NAMES nvshmem
               HINTS $ENV{NVSHMEM_HOME} $ENV{NVSHMEM_ROOT}
               PATH_SUFFIXES lib lib64)
  if(NOT QCU_NVSHMEM_INCLUDE_DIR OR
     NOT QCU_NVSHMEMX_INCLUDE_DIR OR
     NOT QCU_NVSHMEM_LIBRARY)
    message(FATAL_ERROR
      "QCU_ENABLE_NVSHMEM=ON requires nvshmem.h, nvshmemx.h and libnvshmem")
  endif()
  target_include_directories(qcu PRIVATE
    ${QCU_NVSHMEM_INCLUDE_DIR} ${QCU_NVSHMEMX_INCLUDE_DIR})
  target_compile_definitions(qcu PRIVATE QCU_HAVE_NVSHMEM=1)
  target_link_libraries(qcu ${QCU_NVSHMEM_LIBRARY})
  message(STATUS "NVSHMEM enabled: ${QCU_NVSHMEM_LIBRARY}")
else()
  message(STATUS "NVSHMEM disabled (set -DQCU_ENABLE_NVSHMEM=ON to enable)")
endif()
